cmake_minimum_required(VERSION 3.22.1)
project(PresageVitals CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if (NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif ()

# installed by `apt install libsmartspectra-dev` (see setup.sh)
find_package(SmartSpectra CONFIG REQUIRED)
add_executable(presage_vitals presage_vitals.cpp)
target_link_libraries(presage_vitals SmartSpectra::SDK)
